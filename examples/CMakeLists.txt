# Extra arguments name the library to link; the default is the base SDK.
function(alpaca_add_example tgt)
    set(libs ${ARGN})
    if (NOT libs)
        set(libs slick::alpaca-cpp)
    endif()
    add_executable(${tgt} ${tgt}.cpp)
    target_link_libraries(${tgt} PRIVATE ${libs})
    alpaca_set_target_options(${tgt})
    set_target_properties(${tgt} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/examples)
endfunction()

if (BUILD_ALPACA_CORE)
    alpaca_add_example(trading_overview)
    alpaca_add_example(place_and_cancel_order)
    alpaca_add_example(market_data_overview)
    alpaca_add_example(stream_market_data)
endif()

if (BUILD_ALPACA_OPTIONS_STREAMING)
    alpaca_add_example(stream_option_data slick::alpaca-options-streaming-cpp)
endif()
